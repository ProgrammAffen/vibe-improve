cmake_minimum_required(VERSION 3.21)
project(VibeWhisper LANGUAGES C CXX)

include(CheckIncludeFileCXX)
check_include_file_cxx(array VIBE_HAS_CXX_HEADERS)
if(NOT VIBE_HAS_CXX_HEADERS)
    execute_process(COMMAND xcrun --sdk macosx --show-sdk-path OUTPUT_VARIABLE macos_sdk OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    include_directories(SYSTEM "${macos_sdk}/usr/include/c++/v1")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "")
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "")
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "")
set(GGML_NATIVE OFF CACHE BOOL "")
set(GGML_OPENMP OFF CACHE BOOL "")
set(GGML_ACCELERATE ON CACHE BOOL "")
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "")
if(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64")
    set(GGML_METAL OFF CACHE BOOL "")
    set(GGML_AVX2 OFF CACHE BOOL "")
    set(GGML_FMA OFF CACHE BOOL "")
    set(GGML_F16C OFF CACHE BOOL "")
endif()

add_subdirectory("${WHISPER_SOURCE_DIR}" whisper EXCLUDE_FROM_ALL)
add_executable(vibe-whisper whisper-worker.cpp)
target_compile_features(vibe-whisper PRIVATE cxx_std_17)
target_link_libraries(vibe-whisper PRIVATE whisper)
